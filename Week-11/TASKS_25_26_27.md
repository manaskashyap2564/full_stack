## 📝 Task 26: Create HelloWorld Component

### Step 1: Create a HelloWorld component

Create file: `src/components/HelloWorld.jsx`

```jsx
function HelloWorld() {
  return (
    <div style={{
      textAlign: 'center',
      padding: '50px',
      backgroundColor: '#667eea',
      color: 'white',
      borderRadius: '10px',
      margin: '20px'
    }}>
      <h1>👋 Hello World!</h1>
      <p>This is my first React component</p>
    </div>
  );
}

export default HelloWorld;
```

### Step 2: Import in App.jsx

```jsx
import HelloWorld from './components/HelloWorld';

function App() {
  return (
    <>
      <HelloWorld />
    </>
  );
}

export default App;
```

## 🚀 Task 27: JSX Expressions with Variables & Functions

### Create: `src/components/Greeting.jsx`

```jsx
function Greeting() {
  // Variables
  const name = "Student";
  const currentTime = new Date().toLocaleTimeString();
  const currentDate = new Date().toLocaleDateString();
  
  // Function
  const getGreeting = () => {
    const hour = new Date().getHours();
    if (hour < 12) return "Good Morning";
    if (hour < 18) return "Good Afternoon";
    return "Good Evening";
  };

  // Inline expression
  const isWeekend = new Date().getDay() === 0 || new Date().getDay() === 6;

  return (
    <div style={{
      backgroundColor: '#f8f9fa',
      padding: '30px',
      borderRadius: '15px',
      margin: '20px',
      boxShadow: '0 4px 6px rgba(0,0,0,0.1)'
    }}>
      <h1>{getGreeting()}, {name}! 👋</h1>
      <p>📅 Today is: {currentDate}</p>
      <p>🕐 Current time: {currentTime}</p>
      <p>
        {isWeekend ? '🎉 It\'s the weekend!' : '💼 It\'s a weekday!'}
      </p>
      
      {/* Conditional rendering */}
      {isWeekend && <p style={{color: 'green'}}>Enjoy your weekend!</p>}
      
      {/* Expression in style */}
      <div style={{
        backgroundColor: isWeekend ? '#d4edda' : '#fff3cd',
        padding: '15px',
        borderRadius: '8px',
        marginTop: '15px'
      }}>
        <h3>Math in JSX:</h3>
        <p>2 + 2 = {2 + 2}</p>
        <p>10 * 5 = {10 * 5}</p>
        <p>Random number: {Math.floor(Math.random() * 100)}</p>
      </div>
    </div>
  );
}

export default Greeting;
```

### Update App.jsx:

```jsx
import HelloWorld from './components/HelloWorld';
```
